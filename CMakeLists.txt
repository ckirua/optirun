cmake_minimum_required(VERSION 4.0)
project(hybrid_python VERSION 0.1.0 LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "Milestone 1 supports Linux/POSIX only")
endif()
option(HYBRID_PYTHON_BUILD_TESTS "Build native and Python tests" OFF)
set(HYBRID_PYTHON_GIL_EXECUTABLE "" CACHE FILEPATH "Regular CPython 3.14 executable used by tests")
set(HYBRID_PYTHON_FREE_THREADED_EXECUTABLE "" CACHE FILEPATH "Free-threaded CPython 3.14 executable used by tests")
set(HYBRID_PYTHON_BINDING_EXECUTABLE "" CACHE FILEPATH "Python ABI used to build _native")
if(HYBRID_PYTHON_BINDING_EXECUTABLE)
  set(Python_EXECUTABLE "${HYBRID_PYTHON_BINDING_EXECUTABLE}")
endif()
set(Python_FIND_STRATEGY LOCATION)
find_package(Python 3.14 EXACT COMPONENTS Interpreter Development.Module REQUIRED)
find_package(Threads REQUIRED)
find_package(pybind11 CONFIG REQUIRED)

add_library(hybrid_python STATIC src/runtime_manager.cpp src/wire_codec.cpp src/pipe_transport.cpp)
target_include_directories(hybrid_python PUBLIC include)
target_compile_features(hybrid_python PUBLIC cxx_std_20)
target_link_libraries(hybrid_python PUBLIC Threads::Threads)
target_compile_options(hybrid_python PRIVATE $<$<CXX_COMPILER_ID:GNU,Clang>:-Wall -Wextra -Wpedantic -Werror>)
target_compile_definitions(hybrid_python PRIVATE HYBRID_WORKER_SCRIPT="${CMAKE_CURRENT_SOURCE_DIR}/workers/hybrid_python_worker.py")
set_target_properties(hybrid_python PROPERTIES POSITION_INDEPENDENT_CODE ON)

pybind11_add_module(_native MODULE src/python_bindings.cpp)
target_link_libraries(_native PRIVATE hybrid_python)
target_compile_features(_native PRIVATE cxx_std_20)
set_target_properties(_native PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/python/hybrid_python")
file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/python/hybrid_python")
foreach(file __init__.py __init__.pyi py.typed)
  configure_file("python/hybrid_python/${file}" "${CMAKE_BINARY_DIR}/python/hybrid_python/${file}" COPYONLY)
endforeach()
configure_file(workers/hybrid_python_worker.py "${CMAKE_BINARY_DIR}/python/hybrid_python/worker.py" COPYONLY)
install(TARGETS _native LIBRARY DESTINATION hybrid_python)
install(FILES python/hybrid_python/__init__.py python/hybrid_python/__init__.pyi python/hybrid_python/py.typed DESTINATION hybrid_python)
install(PROGRAMS workers/hybrid_python_worker.py DESTINATION hybrid_python RENAME worker.py)

if(HYBRID_PYTHON_BUILD_TESTS)
  foreach(executable IN ITEMS HYBRID_PYTHON_GIL_EXECUTABLE HYBRID_PYTHON_FREE_THREADED_EXECUTABLE)
    if(NOT ${executable} OR NOT EXISTS "${${executable}}" OR IS_DIRECTORY "${${executable}}")
      message(FATAL_ERROR "${executable} must name an existing Python executable when tests are enabled")
    endif()
  endforeach()
  enable_testing()
  add_executable(native_dispatch examples/native_dispatch.cpp)
  target_link_libraries(native_dispatch PRIVATE hybrid_python)
  add_executable(runtime_manager_test tests/runtime_manager_test.cpp)
  target_link_libraries(runtime_manager_test PRIVATE hybrid_python)
  add_test(NAME runtime_manager_test COMMAND runtime_manager_test "${HYBRID_PYTHON_GIL_EXECUTABLE}" "${HYBRID_PYTHON_FREE_THREADED_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/handlers.py")
  add_test(NAME python_binding_test COMMAND "${Python_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/python_binding_test.py" "${HYBRID_PYTHON_GIL_EXECUTABLE}" "${HYBRID_PYTHON_FREE_THREADED_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/handlers.py")
  set_tests_properties(python_binding_test PROPERTIES ENVIRONMENT "PYTHONPATH=${CMAKE_BINARY_DIR}/python")
endif()
